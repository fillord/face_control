       function sortEmployees(key) {
         if (empSortKey === key) {
           empSortAsc = !empSortAsc;
         } else {
           empSortKey = key;
           empSortAsc = true;
         }
         allEmployees.sort((a, b) => {
           let av, bv;
           if (key === 'name') { av = (a.name || '').toLowerCase(); bv = (b.name || '').toLowerCase(); }
           else if (key === 'dept') {
             const da = allDepts.find(d => d.id === a.dept_id);
             const db = allDepts.find(d => d.id === b.dept_id);
             av = da ? da.name.toLowerCase() : '';
             bv = db ? db.name.toLowerCase() : '';
           }
           else if (key === 'date') { av = a.registered_at || ''; bv = b.registered_at || ''; }
           if (av < bv) return empSortAsc ? -1 : 1;
           if (av > bv) return empSortAsc ? 1 : -1;
           return 0;
         });
         renderEmployees();
         // Update sort arrows
         ['emp-th-name', 'emp-th-dept', 'emp-th-date'].forEach(id => {
           const el = document.getElementById(id);
           if (!el) return;
           const arrow = el.querySelector('.sort-arrow');
           if (arrow) arrow.textContent = '';
         });
         const activeId = key === 'name' ? 'emp-th-name' : key === 'dept' ? 'emp-th-dept' : 'emp-th-date';
         const active = document.getElementById(activeId);
         if (active) {
           const arrow = active.querySelector('.sort-arrow');
           if (arrow) arrow.textContent = empSortAsc ? ' ▲' : ' ▼';
         }
       }

       function sortUsers(key) {
         const tbody = document.getElementById('usersTableBody');
         if (!tbody) return;
         const rows = Array.from(tbody.querySelectorAll('tr[data-username]'));
         if (!rows.length) return;
         if (userSortKey === key) {
           userSortAsc = !userSortAsc;
         } else {
           userSortKey = key;
           userSortAsc = true;
         }
         rows.sort((a, b) => {
           const av = (a.dataset[key] || '').toLowerCase();
           const bv = (b.dataset[key] || '').toLowerCase();
           if (av < bv) return userSortAsc ? -1 : 1;
           if (av > bv) return userSortAsc ? 1 : -1;
           return 0;
         });
         rows.forEach(r => tbody.appendChild(r));
         // Update sort arrows
         ['user-th-username', 'user-th-role'].forEach(id => {
           const el = document.getElementById(id);
           if (!el) return;
           const arrow = el.querySelector('.sort-arrow');
           if (arrow) arrow.textContent = '';
         });
         const activeId = key === 'username' ? 'user-th-username' : 'user-th-role';
         const active = document.getElementById(activeId);
         if (active) {
           const arrow = active.querySelector('.sort-arrow');
           if (arrow) arrow.textContent = userSortAsc ? ' ▲' : ' ▼';
         }
       }
